onmouseover="this.src='./2.jpg'"
onmouseout="this.src='./1.jpg'">


import React, { useState } from 'react';

export default function ImageHoverEffect() {
  // 1. 현재 표시할 이미지 경로를 상태로 관리합니다.
  const [imgSrc, setImgSrc] = useState('./1.jpg');

  return (
    <img
      src={imgSrc}
      alt="이미지"
      // 마우스를 올렸을 때 이미지 변경
      onMouseEnter={() => setImgSrc('./2.jpg')}
      // 마우스가 떠났을 때 원본 이미지로 복구
      onMouseLeave={() => setImgSrc('./1.jpg')}
    />
  );
}
